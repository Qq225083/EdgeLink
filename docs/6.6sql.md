-- 1.1 PLC 设备主表
CREATE TABLE IF NOT EXISTS plc_device (
    id              BIGINT        NOT NULL AUTO_INCREMENT  COMMENT '设备ID',
    device_name     VARCHAR(100)  NOT NULL                 COMMENT '设备名称',
    device_code     VARCHAR(50)   DEFAULT NULL             COMMENT '设备编号（如PLC-Q-01）',
    plc_brand       VARCHAR(50)   NOT NULL DEFAULT 'Mitsubishi' COMMENT 'PLC品牌',
    plc_series      VARCHAR(50)   DEFAULT NULL             COMMENT 'PLC系列（Q/L/FX/iQ-R）',
    com_type        VARCHAR(50)   DEFAULT NULL             COMMENT '通信方式（MC_Protocol/Modbus_TCP/GOT/PLC_RS232C）',
    plc_ip          VARCHAR(50)   NOT NULL                 COMMENT 'PLC IP地址',
    mes_ip          VARCHAR(50)   DEFAULT NULL             COMMENT 'MES/MDPS对接IP',
    mes_port        INT           NOT NULL DEFAULT 0       COMMENT 'MES/MDPS对接端口',
    plc_port        INT           NOT NULL DEFAULT 5007    COMMENT 'PLC端口号',
    mc_frame        VARCHAR(10)   DEFAULT NULL             COMMENT 'MC协议帧格式（3E/4E）',
    station_no      INT           NOT NULL DEFAULT 0       COMMENT '站号',
    network_no      INT           NOT NULL DEFAULT 0       COMMENT '网络号',
    scan_interval_ms INT          NOT NULL DEFAULT 1000    COMMENT '采集周期（毫秒）',
    comm_timeout_ms INT           NOT NULL DEFAULT 3000    COMMENT '通信超时（毫秒）',
    retry_count     INT           NOT NULL DEFAULT 2       COMMENT '失败重试次数',
    retry_interval_ms INT         NOT NULL DEFAULT 500     COMMENT '重试间隔（毫秒）',
    trigger_kind    INT           NOT NULL DEFAULT 0       COMMENT '触发方式（0=握手 1=固定周期 2=变化触发）',
    status          CHAR(1)       NOT NULL DEFAULT '0'     COMMENT '状态（0启用 1停用）',
    create_by       VARCHAR(64)   DEFAULT NULL             COMMENT '创建者',
    create_time     DATETIME      DEFAULT NULL             COMMENT '创建时间',
    update_by       VARCHAR(64)   DEFAULT NULL             COMMENT '更新者',
    update_time     DATETIME      DEFAULT NULL             COMMENT '更新时间',
    remark          VARCHAR(500)  DEFAULT NULL             COMMENT '备注',
    del_flag        CHAR(1)       NOT NULL DEFAULT '0'     COMMENT '删除标志（0正常 2删除）',
    PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='PLC设备表';

-- 1.2 PLC 采集点位从表
CREATE TABLE IF NOT EXISTS plc_tag (
    id               BIGINT       NOT NULL AUTO_INCREMENT  COMMENT '点位ID',
    device_id        BIGINT       NOT NULL                 COMMENT '所属设备ID',
    tag_name         VARCHAR(100) NOT NULL                 COMMENT '点位名称',
    register_type    VARCHAR(10)  NOT NULL                 COMMENT '寄存器类型（D/W/X/Y/M）',
    register_address VARCHAR(50)  NOT NULL                 COMMENT '寄存器地址',
    data_type        VARCHAR(20)  NOT NULL                 COMMENT '数据类型（INT16/INT32/FLOAT/BIT）',
    unit             VARCHAR(20)  DEFAULT NULL             COMMENT '单位',
    description      TEXT         DEFAULT NULL             COMMENT '点位描述',
    status           CHAR(1)      NOT NULL DEFAULT '0'     COMMENT '状态（0启用 1停用）',
    sort_order       INT          NOT NULL DEFAULT 0       COMMENT '排序号',
    create_by        VARCHAR(64)  DEFAULT NULL             COMMENT '创建者',
    create_time      DATETIME     DEFAULT NULL             COMMENT '创建时间',
    update_by        VARCHAR(64)  DEFAULT NULL             COMMENT '更新者',
    update_time      DATETIME     DEFAULT NULL             COMMENT '更新时间',
    del_flag         CHAR(1)      NOT NULL DEFAULT '0'     COMMENT '删除标志（0正常 2删除）',
    PRIMARY KEY (id),
    CONSTRAINT fk_plc_tag_device FOREIGN KEY (device_id) REFERENCES plc_device (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='PLC采集点位表';


-- ============================================================
-- 二、ALTER TABLE（在已有 plc_device / plc_tag 上补充字段）
--   如果表是全新创建的，跳过此段
-- ============================================================

-- 2.1 plc_device 新增字段
ALTER TABLE plc_device ADD COLUMN device_code VARCHAR(50) DEFAULT NULL COMMENT '设备编号（如PLC-Q-01）' AFTER device_name;
ALTER TABLE plc_device ADD COLUMN mes_ip VARCHAR(50) DEFAULT NULL COMMENT 'MES/MDPS对接IP' AFTER plc_ip;
ALTER TABLE plc_device ADD COLUMN mes_port INT NOT NULL DEFAULT 0 COMMENT 'MES/MDPS对接端口' AFTER mes_ip;
ALTER TABLE plc_device ADD COLUMN comm_timeout_ms INT NOT NULL DEFAULT 3000 COMMENT '通信超时（毫秒）' AFTER scan_interval_ms;
ALTER TABLE plc_device ADD COLUMN retry_count INT NOT NULL DEFAULT 2 COMMENT '失败重试次数' AFTER comm_timeout_ms;
ALTER TABLE plc_device ADD COLUMN retry_interval_ms INT NOT NULL DEFAULT 500 COMMENT '重试间隔（毫秒）' AFTER retry_count;
ALTER TABLE plc_device ADD COLUMN del_flag CHAR(1) NOT NULL DEFAULT '0' COMMENT '删除标志（0正常 2删除）' AFTER remark;

-- 2.2 plc_tag 新增字段（含 enabled→status 重命名）
ALTER TABLE plc_tag CHANGE COLUMN enabled status CHAR(1) NOT NULL DEFAULT '0' COMMENT '状态（0启用 1停用）';
ALTER TABLE plc_tag ADD COLUMN create_by VARCHAR(64) DEFAULT NULL COMMENT '创建者' AFTER sort_order;
ALTER TABLE plc_tag ADD COLUMN create_time DATETIME DEFAULT NULL COMMENT '创建时间' AFTER create_by;
ALTER TABLE plc_tag ADD COLUMN update_by VARCHAR(64) DEFAULT NULL COMMENT '更新者' AFTER create_time;
ALTER TABLE plc_tag ADD COLUMN update_time DATETIME DEFAULT NULL COMMENT '更新时间' AFTER update_by;
ALTER TABLE plc_tag ADD COLUMN del_flag CHAR(1) NOT NULL DEFAULT '0' COMMENT '删除标志（0正常 2删除）' AFTER update_time;

-- 2.3 去掉 FK 的 ON DELETE CASCADE
ALTER TABLE plc_tag DROP FOREIGN KEY fk_plc_tag_device;
ALTER TABLE plc_tag ADD CONSTRAINT fk_plc_tag_device FOREIGN KEY (device_id) REFERENCES plc_device (id);

-- 3.2 点位表（最关键）
CREATE INDEX idx_tag_device_id ON plc_tag(device_id);
CREATE INDEX idx_tag_status_del ON plc_tag(status, del_flag);
CREATE INDEX idx_tag_device_status ON plc_tag(device_id, status, del_flag);
